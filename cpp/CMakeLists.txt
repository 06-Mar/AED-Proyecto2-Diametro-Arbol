cmake_minimum_required(VERSION 3.16)
project(DiametroArbol LANGUAGES CXX)
add_executable(diametro src/main.cpp src/tree.cpp src/bfs.cpp src/diameter.cpp src/json_export.cpp)
target_include_directories(diametro PRIVATE include)
target_compile_features(diametro PRIVATE cxx_std_17)
set_target_properties(diametro PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(diametro PRIVATE /W4 /permissive-)
else()
    target_compile_options(diametro PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME contrato_y_algoritmo COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_diametro.py" "$<TARGET_FILE:diametro>")
    endif()
endif()
